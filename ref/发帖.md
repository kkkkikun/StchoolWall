## 3. 逻辑层：`pages/publish/publish.js`

JavaScript

```
// pages/publish/publish.js
const http = require('../../utils/request.js');

Page({
  data: {
    hotTags: ['考研交流', '失物招领', '二手交易', '拼车出行'],
    selectedSection: '',       // 已选择的版块名称
    content: '',               // 帖子文本正文
    uploadedImages: [],        // 存储微信临时图片路径数组
    uploadedVideo: null,       // 存储微信临时视频路径
    aiEnabled: false           // AI辅助状态
  },

  // 监听正文输入
  onContentInput(e) {
    this.setData({ content: e.detail.value });
  },

  // 快捷点击热搜标签追加到文本末尾
  handleTagTap(e) {
    const { tag } = e.currentTarget.dataset;
    const currentContent = this.data.content;
    this.setData({
      content: currentContent + ` #${tag} `
    });
  },

  // 打开版块选择半弹窗或跳转二级页面
  handleSelectSection() {
    // 留给 Claude Code 去实现具体选择器逻辑
    wx.navigateTo({
      url: '/pages/publish/section-picker',
      events: {
        // 监听下级页面选择版块后的回调
        sectionSelected: (data) => {
          this.setData({ selectedSection: data.name });
        }
      }
    });
  },

  // 处理侧边栏快捷操作
  onToolbarAction(e) {
    const { type } = e.currentTarget.dataset;
    console.log(`触发了快捷工具: ${type}`);
    // 比如地理位置处理：
    if (type === 'location') {
      wx.getLocation({
        type: 'wgs84',
        success: (res) => { console.log('当前位置：', res); }
      });
    }
  },

  // 处理多媒体资产上传
  handleUploadMedia(e) {
    const { type } = e.currentTarget.dataset;
    if (type === 'image') {
      wx.chooseMedia({
        count: 9,
        mediaType: ['image'],
        sourceType: ['album', 'camera'],
        success: (res) => {
          const paths = res.tempFiles.map(file => file.tempFilePath);
          this.setData({ uploadedImages: [...this.data.uploadedImages, ...paths] });
        }
      });
    } else if (type === 'video') {
      wx.chooseMedia({
        count: 1,
        mediaType: ['video'],
        sourceType: ['album', 'camera'],
        success: (res) => {
          this.setData({ uploadedVideo: res.tempFiles[0].tempFilePath });
        }
      });
    }
  },

  // 监听AI协助开关状态
  onAiSwitchChange(e) {
    this.setData({ aiEnabled: e.detail.value });
    if (e.detail.value && this.data.content) {
      this.triggerAiOptimize();
    }
  },

  // 模拟请求后端大模型润色文本
  async triggerAiOptimize() {
    wx.showLoading({ title: 'AI 优化中...' });
    try {
      // 留给 Claude Code 接入大模型后端接口的标准架构
      // const res = await http.post('/api/v1/ai/optimize', { content: this.data.content });
      // this.setData({ content: res.optimizedContent });
    } catch (err) {
      console.error('AI优化失败', err);
    } finally {
      wx.hideLoading();
    }
  },

  // 提交发布表单 (核心分离骨架)
  async handleSubmitPost() {
    const { content, selectedSection, aiEnabled } = this.data;
    if (!selectedSection) {
      return wx.showToast({ title: '请选择发布的版块', icon: 'none' });
    }
    if (!content.trim()) {
      return wx.showToast({ title: '请填写一点内容吧', icon: 'none' });
    }

    wx.showLoading({ title: '发布中...', mask: true });

    try {
      // 1. 如果有媒体文件，通常需要先调用上传文件接口
      // 2. 组合最终表单请求后端接口
      const payload = {
        section: selectedSection,
        content: content,
        isAiOptimized: aiEnabled
      };
      
      // const response = await http.post('/api/v1/posts/create', payload);
      
      wx.hideLoading();
      wx.showToast({ title: '发布成功', icon: 'success' });
      
      // 延迟返回首页
      setTimeout(() => { wx.switchTab({ url: '/pages/index/index' }); }, 1500);
    } catch (err) {
      wx.hideLoading();
      console.error('发布失败', err);
    }
  }
});
```
