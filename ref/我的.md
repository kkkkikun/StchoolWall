// pages/mine/mine.js
const http = require('../../utils/request.js');

Page({
  data: {
    // 严密复刻高保真图示数据
    userInfo: {
      avatarUrl: '/assets/images/avatar_eagle.png', // 对应图中的蓝色小鹰头像
      nickName: '鄂小鹰0001',
      schoolName: '鄂尔多斯应用技术学院'
    },

    // 统计面板真实映射
    userStats: {
      postsCount: 124,
      commentsCount: 231,
      favoritesCount: 86,
      footprintsCount: 45
    },

    // 分组项数据定义
    menuGroup1: [
      { name: '社区规范', icon: '/assets/icons/menu_shield.png', action: 'guidelines' },
      { name: '新手帮助', icon: '/assets/icons/menu_help.png', action: 'help' },
      { name: '邀请校友', icon: '/assets/icons/menu_invite.png', action: 'invite' }
    ],

    menuGroup2: [
      { name: '设置', icon: '/assets/icons/menu_setting.png', action: 'settings' },
      { name: '关于我们', icon: '/assets/icons/menu_about.png', action: 'about' },
      { name: '联系我们', icon: '/assets/icons/menu_contact.png', action: 'contact' }
    ]
  },

  onShow() {
    // 每次切回个人中心，更新最新的数据资产数值
    this.fetchUserDashboardData();
  },

  // 下拉同步拉取
  async onPullDownRefresh() {
    await this.fetchUserDashboardData();
    wx.stopPullDownRefresh();
  },

  // 分离式服务端数据同步骨架
  async fetchUserDashboardData() {
    try {
      // 留给 Claude Code 接入前后端分离联调的请求模版
      // const profile = await http.get('/api/v1/user/profile');
      // const stats = await http.get('/api/v1/user/stats');
      // this.setData({ userInfo: profile, userStats: stats });
    } catch (err) {
      console.error('更新个人中心资产失败', err);
    }
  },

  // 四列核心数字点击跳转二级细分列表（如我的发布列表、我的收藏列表等）
  navigateToStats(e) {
    const { type } = e.currentTarget.dataset;
    wx.navigateTo({
      url: `/pages/mine/my-records/my-records?type=${type}`
    });
  },

  // 统一分发列表卡片的点击路由行为
  handleMenuTap(e) {
    const { action } = e.currentTarget.dataset;
    console.log(`用户点击了操作项: ${action}`);
    
    switch (action) {
      case 'guidelines':
        wx.navigateTo({ url: '/pages/webview/guidelines' });
        break;
      case 'help':
        wx.navigateTo({ url: '/pages/help/help' });
        break;
      case 'invite':
        // 触发微信原生分享或海报弹窗
        break;
      case 'settings':
        wx.navigateTo({ url: '/pages/settings/settings' });
        break;
      case 'about':
        wx.navigateTo({ url: '/pages/about/about' });
        break;
      case 'contact':
        // 可以直接调用客服会话，或者反馈页面
        wx.navigateTo({ url: '/pages/contact/feedback' });
        break;
      default:
        break;
    }
  }
});
