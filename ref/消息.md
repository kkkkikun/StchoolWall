```
      </view>
      <text class="category-label">{{item.name}}</text>
    </view>
  </view>

  <!-- 3. 帖子动态与全量操作栏 -->
  <view class="dynamic-control-bar flex-row space-between">
    <text class="section-title">帖子动态</text>
    <view class="mark-read-btn flex-row" bindtap="handleMarkAllRead">
      <text>全部已读</text>
      <!-- 带有灰色对勾小图标 -->
      <icon class="icon-check" type="success_no_circle" size="12" color="#999999" />
    </view>
  </view>

  <!-- 4. 动态内容展示区（条件渲染：处理空状态与真实列表） -->
  <view class="message-content-area">
    
    <!-- 状态 A：无消息流（严格复刻当前截图效果） -->
    <block wx:if="{{dynamicMessages.length === 0}}">
      <view class="empty-state flex-col">
        <text class="empty-text">暂无消息</text>
      </view>
    </block>
    
    <!-- 状态 B：有消息流（留给 Claude Code 编写具体的二级样式与动态渲染） -->
    <block wx:else>
      <view class="message-card flex-row" wx:for="{{dynamicMessages}}" wx:key="id" bindtap="navigateToPost" data-post-id="{{item.postId}}">
        <image class="sender-avatar" src="{{item.avatar}}" mode="aspectFill" />
        <view class="msg-main flex-col">
          <view class="msg-top flex-row space-between">
            <text class="sender-name">{{item.nickname}}</text>
            <text class="msg-time">{{item.time}}</text>
          </view>
          <text class="msg-text">{{item.brief}}</text>
        </view>
      </view>
    </block>

  </view>

</view>
```

```
// pages/message/message.js
const http = require('../../utils/request.js');

Page({
  data: {
    // 对应顶部的四个圆形大分类通知，其中私信初始带12条未读
    categories: [
      { type: 'comment', name: '评论', icon: '/assets/icons/msg_comment.png', colorClass: 'blue', unreadCount: 0 },
      { type: 'like', name: '赞', icon: '/assets/icons/msg_like.png', colorClass: 'pink', unreadCount: 0 },
      { type: 'party', name: '组局', icon: '/assets/icons/msg_party.png', colorClass: 'yellow', unreadCount: 0 },
      { type: 'im', name: '私信', icon: '/assets/icons/msg_im.png', colorClass: 'green', unreadCount: 12 }
    ],
    
    // 帖子动态列表数据，默认为空以渲染“暂无消息”
    dynamicMessages: []
  },

  onShow() {
    // 消息页面通常切换 Tab 时即需刷新未读数，建议放在 onShow 中
    this.fetchUnreadCounts();
  },

  // 下拉刷新全局消息状态
  async onPullDownRefresh() {
    await this.fetchUnreadCounts();
    wx.stopPullDownRefresh();
  },

  // 分离式后端请求骨架
  async fetchUnreadCounts() {
    try {
      // 留给 Claude Code 接入前后端分离 API 的调用模版
      // const res = await http.get('/api/v1/message/unread-summary');
      // this.setData({
      //   'categories[0].unreadCount': res.commentCount,
      //   'categories[1].unreadCount': res.likeCount,
      //   'categories[2].unreadCount': res.partyCount,
      //   'categories[3].unreadCount': res.imCount,
      //   dynamicMessages: res.recentFeeds
      // });
    } catch (err) {
      console.error('获取消息通知未读数失败', err);
    }
  },

  // 点击“全部已读”动作
  async handleMarkAllRead() {
    // 若当前已经是空状态，无需发起多余的网络请求
    if (this.data.dynamicMessages.length === 0) {
      return wx.showToast({ title: '没有未读动态哦', icon: 'none' });
    }

    wx.showLoading({ title: '正在全量标记...', mask: true });
    try {
      // 调用全量已读接口
      // await http.post('/api/v1/message/mark-all-read');
      this.setData({ dynamicMessages: [] }); // 清空数据触发“暂无消息”渲染
      wx.hideLoading();
      wx.showToast({ title: '清除成功', icon: 'success' });
    } catch (err) {
      wx.hideLoading();
      console.error('清除未读失败', err);
    }
  },

  // 分流跳转对应的二级列表页（评论页、点赞人页、攒局邀请、IM私信对话）
  navigateToCategory(e) {
    const { type } = e.currentTarget.dataset;
    wx.navigateTo({
      url: `/pages/message/sub-list/sub-list?type=${type}`
    });
  },

  // 点击具体帖子动态通知时的跳转路由
  navigateToPost(e) {
    const { postId } = e.currentTarget.dataset;
    wx.navigateTo({
      url: `/pages/post/detail?id=${postId}`
    });
  }
});
```
